        function getDates() {
            const today = new Date();
            const past = new Date();
            past.setDate(today.getDate() - 5); 
            const formatDate = (date) => date.toISOString().split('T')[0];
            return { to: formatDate(today), from: formatDate(past) };
        }

        // --- DUAL API QUEUE ENGINE ---
        const apiQueue = [];
        let isProcessingQueue = false;

        function enqueueAPI(type, payload) {
            return new Promise((resolve) => {
                apiQueue.push({ type, payload, resolve });
                if (!isProcessingQueue) processQueue();
            });
        }

        async function processQueue() {
            isProcessingQueue = true;
            while (apiQueue.length > 0) {
                const { type, payload, resolve } = apiQueue.shift();
                try {
                    if (type === 'finnhub') {
                        const response = await fetch(`https://finnhub.io/api/v1${payload}&token=${API_KEY}`);
                        resolve(response.ok ? await response.json() : null);
                        
                    } else if (type === 'yahoo') {
                        const { symbol, isIntraday } = payload;
                        const interval = isIntraday ? '15m' : '1d';
                        const range = isIntraday ? '5d' : '1y';
                        const cb = Math.floor(Date.now() / 60000);
                        
                        // Added includePrePost=true to pull pre-market metrics
                        const yfUrl = `https://query1.finance.yahoo.com/v8/finance/chart/${symbol}?interval=${interval}&range=${range}&includePrePost=true&cb=${cb}`;
                        const proxyUrl = `https://api.allorigins.win/raw?url=${encodeURIComponent(yfUrl)}`;
                        
                        const response = await fetch(proxyUrl);
                        const data = response.ok ? await response.json() : null;
                        
                        if (data && data.chart && data.chart.result && data.chart.result.length > 0) {
                            const result = data.chart.result[0];
                            const indicators = result.indicators.quote[0];
                            resolve({
                                s: 'ok',
                                t: result.timestamp,
                                c: indicators.close,
                                h: indicators.high,
                                l: indicators.low,
                                v: indicators.volume,
                                meta: result.meta // Exposed meta to access session blocks
                            });
                        } else {
                            resolve(null);
                        }
                    }
                } catch (e) {
                    console.error("API Error:", e);
                    resolve(null);
                }
                await new Promise(r => setTimeout(r, 800)); 
            }
            isProcessingQueue = false;
        }

        async function loadStockData(symbol) {
            const [profile, quote, news, recs, dailyCandles, intradayCandles] = await Promise.all([
                enqueueAPI('finnhub', `/stock/profile2?symbol=${symbol}`),
                enqueueAPI('finnhub', `/quote?symbol=${symbol}`),
                enqueueAPI('finnhub', `/company-news?symbol=${symbol}&from=${dates.from}&to=${dates.to}`),
                enqueueAPI('finnhub', `/stock/recommendation?symbol=${symbol}`),
                enqueueAPI('yahoo', { symbol: symbol, isIntraday: false }),
                enqueueAPI('yahoo', { symbol: symbol, isIntraday: true })
            ]);

            renderCard(symbol, profile, quote, news, recs, dailyCandles, intradayCandles);
            
            loadedSymbols++;
            const progEl = document.getElementById('progressText');
            if (loadedSymbols < totalSymbols) {
                progEl.innerText = `Updating data... (${loadedSymbols}/${totalSymbols})`;
            } else {
                progEl.innerText = `Matrix Synchronized`;
                progEl.classList.remove('animate-pulse');
                progEl.classList.add('text-green-400');
                setTimeout(() => { progEl.innerText = 'Live'; }, 5000);
            }
        }

        function renderCard(symbol, profile, quote, news, recs, dailyCandles, intradayCandles) {
            const currentPrice = quote?.c || 0;
            const openPrice = quote?.o || 0;
            const cardEl = document.getElementById(`card-${symbol}`);
            if (!cardEl) return;

            let shadowClass = 'shadow-yellow';
            let arrow = '⬌';
            let arrowColor = 'text-yellow-500';

            if (currentPrice > openPrice) {
                shadowClass = 'shadow-green';
                arrow = '▲';
                arrowColor = 'text-green-400';
            } else if (currentPrice < openPrice) {
                shadowClass = 'shadow-red';
                arrow = '▼';
                arrowColor = 'text-red-400';
            }

            // --- PRE-MARKET HIGH/LOW/CLOSE EXTRACTION ---
            let preMarketHTML = '';
            if (intradayCandles && intradayCandles.s === 'ok' && intradayCandles.meta) {
                const pre = intradayCandles.meta.currentTradingPeriod?.pre;
                if (pre && pre.start && pre.end) {
                    let preHigh = -Infinity;
                    let preLow = Infinity;
                    let preClose = null;
                    
                    for (let i = 0; i < intradayCandles.t.length; i++) {
                        const time = intradayCandles.t[i];
                        // Filter for only the current day's pre-market session
                        if (time >= pre.start && time < pre.end) {
                            if (intradayCandles.h[i] > preHigh) preHigh = intradayCandles.h[i];
                            if (intradayCandles.l[i] < preLow) preLow = intradayCandles.l[i];
                            if (intradayCandles.c[i] !== null && !isNaN(intradayCandles.c[i])) {
                                preClose = intradayCandles.c[i];
                            }
                        }
                    }
                    
                    if (preClose !== null && preHigh !== -Infinity && preLow !== Infinity) {
                        preMarketHTML = `
                            <div class="flex items-center justify-between text-xs mb-4 bg-slate-900/60 p-2.5 rounded-lg border border-slate-700/50 w-full">
                                <span class="text-slate-400 uppercase tracking-widest text-[10px] font-bold">Pre-Market <span class="text-[9px] text-slate-500 normal-case">(H/L/C)</span></span>
                                <div class="flex items-center space-x-3 font-medium">
                                    <span class="text-[10px] text-slate-500">H: <span class="text-slate-300">$${preHigh.toFixed(2)}</span></span>
                                    <span class="text-[10px] text-slate-500">L: <span class="text-slate-300">$${preLow.toFixed(2)}</span></span>
                                    <span class="text-[10px] text-slate-500">C: <span class="text-cyan-400 font-bold">$${preClose.toFixed(2)}</span></span>
                                </div>
                            </div>
                        `;
                    }
                }
            }

            // --- VWAP CALCULATION & DT GO LONG RANGE SIGNAL ---
            let vwapHTML = `<span class="text-slate-500 font-normal ml-1">N/A</span>`;
            let dtSignalHTML = '';

            if (intradayCandles && intradayCandles.s === 'ok' && intradayCandles.t && intradayCandles.t.length > 0) {
                const regular = intradayCandles.meta?.currentTradingPeriod?.regular;
                let cumVolPrice = 0;
                let cumVol = 0;

                if (regular && regular.start && regular.end) {
                    // Isolate today's regular trading hours for standard VWAP
                    for (let i = 0; i < intradayCandles.t.length; i++) {
                        const time = intradayCandles.t[i];
                        if (time >= regular.start && time < regular.end) {
                            const typicalPrice = (intradayCandles.h[i] + intradayCandles.l[i] + intradayCandles.c[i]) / 3;
                            const volume = intradayCandles.v[i];
                            if (volume > 0) {
                                cumVolPrice += typicalPrice * volume;
                                cumVol += volume;
                            }
                        }
                    }
                } else {
                    // Fallback
                    const lastCandleDate = new Date(intradayCandles.t[intradayCandles.t.length - 1] * 1000).toDateString();
                    const currentDayStartIndex = intradayCandles.t.findIndex(time => new Date(time * 1000).toDateString() === lastCandleDate);
                    if (currentDayStartIndex !== -1) {
                         for (let i = currentDayStartIndex; i < intradayCandles.t.length; i++) {
                            const typicalPrice = (intradayCandles.h[i] + intradayCandles.l[i] + intradayCandles.c[i]) / 3;
                            const volume = intradayCandles.v[i];
                            if (volume > 0) {
                                cumVolPrice += typicalPrice * volume;
                                cumVol += volume;
                            }
                        }
                    }
                }
                
                if (cumVol > 0) {
                    const vwap = cumVolPrice / cumVol;
                    const isAbove = currentPrice >= vwap;
                    const vClass = isAbove ? 'text-green-400 bg-green-900/30' : 'text-red-400 bg-red-900/30';
                    const vLabel = isAbove ? '▲ ABOVE' : '▼ BELOW';
                    
                    if (!isAbove) {
                        dtSignalHTML = `<span class="text-[10px] font-extrabold text-amber-400 bg-amber-400/10 px-2 py-0.5 rounded border border-amber-400/30 tracking-wider uppercase animate-pulse">DT Go Long Range</span>`;
                    }

                    vwapHTML = `
                        <span class="text-white font-bold ml-1">$${vwap.toFixed(2)}</span>
                        <span class="ml-2 px-1.5 py-0.5 rounded text-[10px] font-extrabold ${vClass}">${vLabel}</span>
                    `;
                }
            }

            let histHTML = '';
            if (dailyCandles && dailyCandles.s === 'ok' && dailyCandles.c && dailyCandles.c.length > 0) {
                const closes = dailyCandles.c;

                        price = closes[index];

                        }

                        ${n.headline}

            if (!newsHTML) newsHTML = `<li class="text-slate-500 text-xs italic">No recent news available.</li>`;

                <div class="flex items-center space-x-3 mb-3 w-full">
                    <img src="${logoSrc}" alt="${symbol}" class="w-12 h-12 rounded-full bg-white object-contain p-1 border border-slate-600 flex-shrink-0">

                    <div class="text-[10px] font-bold text-slate-400 uppercase tracking-wider mb-2 flex justify-between items-center">
                        <span>Hist. Close vs Open</span>
                        <span class="text-[9px] text-slate-500 font-normal">(5D, 20D, 60D)</span>
                    </div>
                    ${histHTML}

                ${preMarketHTML}

                <!-- Interday VWAP Box with DT Go Long Range indicator -->
                <div class="flex items-center justify-between text-xs mb-4 bg-slate-900/60 p-2.5 rounded-lg border border-slate-700/50 w-full">
